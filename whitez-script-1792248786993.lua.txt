local Config = {
    Enabled = false,
    Speed = 75,
    Min = 16,
    Max = 200,
    Mode = "Physics",
    Particles = true,
}

local FILE = "WhitezMonoSpeedV2.json"
local SaveVersion = 0

local function ValidNumber(value)
    local number = tonumber(value)
    if not number or number ~= number or math.abs(number) == math.huge then
        return nil
    end
    return number
end

local function NormalizeSpeed(value)
    local number = ValidNumber(value)
    if not number then
        return Config.Speed
    end
    return math.clamp(math.floor(number + 0.5), Config.Min, Config.Max)
end

pcall(function()
    if type(readfile) ~= "function" then
        return
    end

    if type(isfile) == "function" and not isfile(FILE) then
        return
    end

    local data = HttpService:JSONDecode(readfile(FILE))

    if type(data) ~= "table" then
        return
    end

    Config.Speed = NormalizeSpeed(data.Speed)

    if data.Mode == "Physics" or data.Mode == "WalkSpeed" then
        Config.Mode = data.Mode
    end

    if type(data.Particles) == "boolean" then
        Config.Particles = data.Particles
    end
end)

local function SaveNow()
    pcall(function()
        if type(writefile) == "function" then
            writefile(FILE, HttpService:JSONEncode({
                Speed = Config.Speed,
                Mode = Config.Mode,
                Particles = Config.Particles,
            }))
        end
    end)
end

local function ScheduleSave()
    SaveVersion += 1
    local version = SaveVersion

    task.delay(0.5, function()
        if Alive and version == SaveVersion then
            SaveNow()
        end
    end)
end

local function New(className, properties, parent)
    local object = Instance.new(className)

    for name, value in pairs(properties or {}) do
        object[name] = value
    end

    object.Parent = parent
    return object
end

local function Corner(parent, radius)
    return New("UICorner", {
        CornerRadius = UDim.new(0, radius or 12),
    }, parent)
end

local function Stroke(parent, transparency)
    return New("UIStroke", {
        Color = Color3.new(1, 1, 1),
        Transparency = transparency or 0.85,
        Thickness = 1,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    }, parent)
end

local function Gradient(parent, top, bottom)
    return New("UIGradient", {
        Rotation = 90,
        Color = ColorSequence.new(top, bottom),
    }, parent)
end

local function Animate(object, properties, duration)
    if not Alive or not object.Parent then
        return
    end

    if Tweens[object] then
        pcall(function()
            Tweens[object]:Cancel()
        end)
    end

    local tween = TweenService:Create(
        object,
        TweenInfo.new(
            duration or 0.18,
            Enum.EasingStyle.Quad,
            Enum.EasingDirection.Out
        ),
        properties
    )

    Tweens[object] = tween
    tween:Play()

    task.delay((duration or 0.18) + 0.1, function()
        if Tweens[object] == tween then
            Tweens[object] = nil
        end
    end)
end

local function ClampPosition(object, x, y)
    local viewport = Viewport()
    local size = object.AbsoluteSize

    return math.clamp(x, 6, math.max(6, viewport.X - size.X - 6)),
        math.clamp(y, 6, math.max(6, viewport.Y - size.Y - 6))
end

local InitializedLayout = false

local function Layout()
    local viewport = Viewport()
    local scale = math.min(
        1,
        (viewport.X - 24) / WIDTH,
        (viewport.Y - 24) / HEIGHT
    )
    scale = math.max(0.1, scale)

    WindowScale.Scale = scale
    FloatingScale.Scale = math.clamp(scale, 0.8, 1)
    ToastScale.Scale = math.max(0.1, math.min(1, (viewport.X - 20) / 310))

    if not InitializedLayout then
        InitializedLayout = true

        Window.Position = UDim2.fromOffset(
            math.max(6, (viewport.X - WIDTH * scale) / 2),
            math.max(6, (viewport.Y - HEIGHT * scale) / 2)
        )

        Floating.Position = UDim2.fromOffset(
            14,
            math.max(6, viewport.Y * 0.5 - 24)
        )
    else
        for _, object in ipairs({Window, Floating}) do
            local x, y = ClampPosition(
                object,
                object.Position.X.Offset,
                object.Position.Y.Offset
            )
            object.Position = UDim2.fromOffset(x, y)
        end
    end
end

local DragStates = {}

local function MakeDraggable(handle, target)
    local state = {
        Active = false,
        Moved = false,
        Input = nil,
        Start = Vector2.zero,
        Origin = Vector2.zero,
    }

    table.insert(DragStates, state)

    Connect(handle.InputBegan, function(input)
        if input.UserInputType ~= Enum.UserInputType.MouseButton1
            and input.UserInputType ~= Enum.UserInputType.Touch then
            return
        end

        if state.Active then
            return
        end

        state.Active = true
        state.Moved = false
        state.Input = input
        state.Start = Vector2.new(input.Position.X, input.Position.Y)
        state.Origin = Vector2.new(
            target.Position.X.Offset,
            target.Position.Y.Offset
        )
    end)

    Connect(UIS.InputChanged, function(input)
        if not state.Active or not state.Input then
            return
        end

        local touch = state.Input.UserInputType == Enum.UserInputType.Touch

        if touch and input ~= state.Input then
            return
        end

        if not touch and input.UserInputType ~= Enum.UserInputType.MouseMovement then
            return
        end

        local delta = Vector2.new(input.Position.X, input.Position.Y) - state.Start

        if delta.Magnitude > 6 then
            state.Moved = true
        end

        if state.Moved then
            local x, y = ClampPosition(
                target,
                state.Origin.X + delta.X,
                state.Origin.Y + delta.Y
            )
            target.Position = UDim2.fromOffset(x, y)
        end
    end)

    Connect(UIS.InputEnded, function(input)
        if input == state.Input
            or (
                state.Input
                and state.Input.UserInputType == Enum.UserInputType.MouseButton1
                and input.UserInputType == Enum.UserInputType.MouseButton1
            ) then
            state.Active = false
            state.Input = nil
        end
    end)

    return state
end

local Character
local Humanoid
local Root
local OriginalWalkSpeed
local BoostApplied = false
local BindingVersion = 0
local LastErrorNotice = 0

local function CharacterReady()
    return Character
        and Character == Player.Character
        and Humanoid
        and Humanoid.Parent
        and Humanoid.Health > 0
        and Root
        and Root.Parent
end

local function PauseReason()
    if not CharacterReady() then
        return "AGUARDANDO PERSONAGEM"
    end

    if Root.Anchored then
        return "PERSONAGEM ANCORADO"
    end

    if Humanoid.Sit then
        return "PAUSADO · SENTADO"
    end

    if Humanoid.PlatformStand then
        return "PAUSADO · SEM CONTROLE"
    end

    local state = Humanoid:GetState()

    if state == Enum.HumanoidStateType.Swimming then
        return "PAUSADO · NADANDO"
    end

    if state == Enum.HumanoidStateType.Climbing then
        return "PAUSADO · ESCALANDO"
    end

    if state == Enum.HumanoidStateType.Physics
        or state == Enum.HumanoidStateType.Ragdoll
        or state == Enum.HumanoidStateType.FallingDown
        or state == Enum.HumanoidStateType.GettingUp then
        return "PAUSADO · ESTADO FÍSICO"
    end

    return nil
end

local function RestoreMovement()
    pcall(function()
        if Humanoid and Humanoid.Parent and OriginalWalkSpeed ~= nil then
            Humanoid.WalkSpeed = OriginalWalkSpeed
        end
    end)

    -- Remove somente o excesso de velocidade horizontal que o
    -- próprio script aplicou. Não altera o componente vertical.
    pcall(function()
        if BoostApplied and Root and Root.Parent and not Root.Anchored then
            local velocity = Root.AssemblyLinearVelocity
            local horizontal = Vector3.new(velocity.X, 0, velocity.Z)
            local limit = math.max(0, OriginalWalkSpeed or 16)

            if horizontal.Magnitude > limit and horizontal.Magnitude > 0 then
                horizontal = horizontal.Unit * limit
                Root.AssemblyLinearVelocity = Vector3.new(
                    horizontal.X,
                    velocity.Y,
                    horizontal.Z
                )
            end
        end
    end)

    BoostApplied = false
    OriginalWalkSpeed = nil
end

local function ApplyMovement()
    if not Config.Enabled then
        return
    end

    if PauseReason() then
        if OriginalWalkSpeed ~= nil or BoostApplied then
            RestoreMovement()
        end
        return
    end

    local success, message = pcall(function()
        if OriginalWalkSpeed == nil then
            OriginalWalkSpeed = Humanoid.WalkSpeed
        end

        -- Mantém a configuração padrão alinhada à velocidade escolhida.
        if Humanoid.WalkSpeed ~= Config.Speed then
            Humanoid.WalkSpeed = Config.Speed
        end

        if Config.Mode ~= "Physics" then
            return
        end

        -- MoveDirection já considera joystick, teclado e controles
        -- normais do Roblox em coordenadas do mundo.
        local direction = Humanoid.MoveDirection
        local horizontal = Vector3.new(direction.X, 0, direction.Z)
        local magnitude = horizontal.Magnitude

        if magnitude > 0.02 then
            local strength = math.min(magnitude, 1)
            local desired = horizontal.Unit * Config.Speed * strength
            local current = Root.AssemblyLinearVelocity

            Root.AssemblyLinearVelocity = Vector3.new(
                desired.X,
                current.Y,
                desired.Z
            )

            BoostApplied = true
        elseif BoostApplied then
            -- Soltar o controle interrompe o impulso em piso normal.
            -- No ar, preserva a inércia do salto.
            if Humanoid.FloorMaterial ~= Enum.Material.Air then
                local current = Root.AssemblyLinearVelocity
                Root.AssemblyLinearVelocity = Vector3.new(0, current.Y, 0)
                BoostApplied = false
            end
        end
    end)

    if not success and os.clock() - LastErrorNotice > 6 then
        LastErrorNotice = os.clock()
        warn("MONO SPEED: " .. tostring(message))
        Notify("Não foi possível aplicar movimento neste personagem.")
    end
end

local function RefreshStatus()
    if not Config.Enabled then
        Status.Text = "DESATIVADO"
        StatusDot.BackgroundColor3 = Color3.fromRGB(115, 115, 125)
        return
    end

    local reason = PauseReason()

    if reason then
        Status.Text = reason
        StatusDot.BackgroundColor3 = Color3.fromRGB(175, 175, 185)
        return
    end

    StatusDot.BackgroundColor3 = Color3.new(1, 1, 1)

    local moving = Humanoid.MoveDirection.Magnitude > 0.02
    local mode = Config.Mode == "Physics" and "FÍSICO" or "WALKSPEED"

    Status.Text = moving
        and (mode .. " · ALVO " .. Config.Speed .. " STUDS/S")
        or (mode .. " · MOVA O PERSONAGEM")
end

local function BindCharacter(character)
    if Character == character then
        return
    end

    BindingVersion += 1
    local version = BindingVersion

    RestoreMovement()
    Character = character
    Humanoid = nil
    Root = nil

    task.spawn(function()
        local deadline = os.clock() + 15

        repeat
            if not Alive
                or version ~= BindingVersion
                or Player.Character ~= character then
                return
            end

            local humanoid = character:FindFirstChildOfClass("Humanoid")
            local root = character:FindFirstChild("HumanoidRootPart")

            if humanoid and root and root:IsA("BasePart") then
                Humanoid = humanoid
                Root = root
                OriginalWalkSpeed = nil
                BoostApplied = false

                ApplyMovement()
                RefreshStatus()
                return
            end

            task.wait(0.1)
        until os.clock() >= deadline

        if Alive and version == BindingVersion then
            Notify("Personagem sem Humanoid ou HumanoidRootPart.")
        end
    end)
end

local function SetEnabled(enabled)
    if Config.Enabled == enabled then
        return
    end

    Config.Enabled = enabled

    if enabled then
        ApplyMovement()
    else
        RestoreMovement()
    end

    PaintSpeed(enabled)
    RefreshStatus()

    if enabled then
        Notify("Ativado. Use o joystick ou WASD para correr.")
    else
        Notify("Desativado. Velocidade original restaurada.")
    end
end

local function RefreshSlider()
    local alpha = (Config.Speed - Config.Min) / (Config.Max - Config.Min)
    Fill.Size = UDim2.fromScale(alpha, 1)
    Knob.Position = UDim2.fromScale(alpha, 0.5)

    if not SpeedInput:IsFocused() then
        SpeedInput.Text = tostring(Config.Speed)
    end
end

local function SetSpeed(value)
    Config.Speed = NormalizeSpeed(value)
    RefreshSlider()
    ApplyMovement()
    RefreshStatus()
    ScheduleSave()
end

local function SetMode(mode)
    RestoreMovement()
    Config.Mode = mode
    Dropdown.Visible = false

    ModeButton.Text = mode == "Physics"
        and "Velocidade física    ▾"
        or "WalkSpeed padrão    ▾"

    ApplyMovement()
    RefreshStatus()
    ScheduleSave()
end

local function SliderFromX(x)
    if Track.AbsoluteSize.X <= 0 then
        return
    end

    local alpha = math.clamp(
        (x - Track.AbsolutePosition.X) / Track.AbsoluteSize.X,
        0,
        1
    )

    SetSpeed(Config.Min + alpha * (Config.Max - Config.Min))
end

local Randomizer = Random.new()
local Particles = {}

for index = 1, 22 do
    local size = Randomizer:NextInteger(1, 3)

    local particle = New("Frame", {
        Name = "Particle" .. index,
        Size = UDim2.fromOffset(size, size),
        BackgroundColor3 = Color3.new(1, 1, 1),
        BackgroundTransparency = 0.65,
        BorderSizePixel = 0,
        ZIndex = 3,
    }, ParticleLayer)
    Corner(particle, 5)

    local data = {
        Object = particle,
        X = Randomizer:NextNumber(0.03, 0.97),
        Y = Randomizer:NextNumber(0, 1),
        Speed = Randomizer:NextNumber(0.02, 0.055),
        Phase = Randomizer:NextNumber(0, math.pi * 2),
        Transparency = Randomizer:NextNumber(0.5, 0.8),
    }

    particle.Position = UDim2.fromScale(data.X, data.Y)
    table.insert(Particles, data)
end

local Elapsed = 0
local ParticleTimer = 0
local StatusTimer = 0
local LayoutTimer = 0
local LastViewport = Viewport()

Connect(RunService.Heartbeat, function(deltaTime)
    if not Alive then
        return
    end

    Elapsed += deltaTime
    ParticleTimer += deltaTime
    StatusTimer += deltaTime
    LayoutTimer += deltaTime

    if StatusTimer >= 0.2 then
        StatusTimer = 0
        RefreshStatus()
    end

    if LayoutTimer >= 0.4 then
        LayoutTimer = 0

        local viewport = Viewport()

        if viewport ~= LastViewport then
            LastViewport = viewport
            Layout()
        end
    end

    if ParticleTimer >= 1 / 30 then
        local step = math.min(ParticleTimer, 0.1)
        ParticleTimer = 0

        if Window.Visible and Config.Particles then
            for _, data in ipairs(Particles) do
                data.Y -= data.Speed * step

                if data.Y < -0.02 then
                    data.Y = 1.02
                    data.X = Randomizer:NextNumber(0.03, 0.97)
                end

                data.Object.Position = UDim2.fromScale(
                    data.X + math.sin(Elapsed * 0.6 + data.Phase) * 0.012,
                    data.Y
                )

                data.Object.BackgroundTransparency = math.clamp(
                    data.Transparency + math.sin(Elapsed + data.Phase) * 0.08,
                    0.35,
                    0.9
                )
            end
        end
    end
end)

Destroy = function()
    if not Alive then
        return
    end

    Alive = false
    BindingVersion += 1
    SaveVersion += 1
    ToastVersion += 1

    SaveNow()
    Config.Enabled = false
    RestoreMovement()

    for _, connection in ipairs(Connections) do
        pcall(function()
            connection:Disconnect()
        end)
    end
    table.clear(Connections)

    for _, tween in pairs(Tweens) do
        pcall(function()
            tween:Cancel()
        end)
    end
    table.clear(Tweens)
    table.clear(Particles)
    table.clear(DragStates)

    pcall(function()
        Gui:Destroy()
    end)

    if Env[KEY] == App then
        Env[KEY] = nil
    end
end

App.Destroy = Destroy

    if Alive and Gui.Parent == nil then

    end
